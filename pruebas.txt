-- Instala la extensión para remover acentos
CREATE EXTENSION IF NOT EXISTS unaccent;

-- Crea una función personalizada y estricta para que PostgreSQL 
-- pueda usar índices rápidos al buscar texto sin acentos
CREATE OR REPLACE FUNCTION unaccent_immutable(text)
RETURNS text AS $$
    SELECT public.unaccent('public.unaccent', $1);
$$ LANGUAGE sql IMMUTABLE STRICT;
